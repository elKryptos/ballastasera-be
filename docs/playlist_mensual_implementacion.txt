QUE HACE
--------
- El admin elige las canciones de cada mes. Los usuarios logueados votan y la
  lista se muestra ordenada por votos.
- Cada cancion pertenece a un mes (columna month_start = primer dia del mes).
  La lista de un mes son sus filas. Los meses anteriores no se borran: son el
  historial que cualquiera puede consultar.
- Mismo patron que los likes de eventos: tabla de votos (una fila por usuario y
  cancion) + contador denormalizado votes_count en la cancion.
- Orden: votes_count DESC, sort_order ASC (el orden del admin desempata),
  title ASC.


REGLAS DECIDIDAS
----------------
1. Lista vigente = el mes mas reciente que tenga canciones y ya haya empezado.
   Si un mes no se carga lista nueva, sigue valiendo la anterior.
2. Se puede cargar un mes con antelacion: no se ve hasta el dia 1.
3. Solo se vota (y se quita el voto) en la lista vigente. Los meses anteriores
   quedan congelados con su ranking final: votar ahi devuelve 409.
4. Solo votan usuarios logueados, un voto por cancion, se puede quitar.
5. La misma cancion puede repetirse otro mes: es otra fila, con votos aparte.
   Dentro de un mismo mes no se puede repetir (UNIQUE month_start + spotify_id).
6. Las canciones se cargan por SQL (como ciudades y estilos). No hay endpoints
   de admin por ahora.
7. No hay tabla "playlists": el mes vive en cada cancion. Solo haria falta si
   la lista del mes tuviera datos propios (titulo, descripcion, portada).


API
---
  GET    /rest/playlist                 publico   canciones de la lista vigente, ordenadas por votos
  GET    /rest/playlist?month=2026-10   publico   las de ese mes (historial); mes sin lista -> []
  GET    /rest/playlist/months          publico   ["2026-11","2026-10",...]; el primero es el vigente
  GET    /rest/playlist/votes           logueado  ids de las canciones de la lista vigente que yo vote
  POST   /rest/playlist/{id}/vote       logueado  204 (idempotente); 409 si no es de la lista vigente
  DELETE /rest/playlist/{id}/vote       logueado  204; 409 si no es de la lista vigente

  Cada cancion: { id, month, danceStyle, title, artist, year, spotifyId, votesCount }
  - month: "2026-10"
  - danceStyle: slug ("salsa" / "bachata"); el FE reparte con el lado A / lado B.
  - month invalido (?month=abc) -> 400.

-- ============================================================================
--  PLAYLIST_TRACKS — canciones de la playlist mensual, elegidas por el admin.
--  Cada fila pertenece a un mes (month_start). La lista de un mes son sus filas
--  y los meses anteriores quedan como historial, no se borra nada.
--  votes_count es un contador denormalizado, igual que events.likes_count.
-- ============================================================================
CREATE TABLE playlist_tracks (
    id             UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    month_start    DATE    NOT NULL,              -- primer dia del mes (2026-11-01 = lista de noviembre)
    dance_style_id BIGINT  NOT NULL REFERENCES dance_styles(id),  -- lado A = Salsa, lado B = Bachata
    title          TEXT    NOT NULL,
    artist         TEXT    NOT NULL,
    release_year   INTEGER,
    spotify_id     TEXT,                          -- id del track en Spotify, para el reproductor
    sort_order     INTEGER NOT NULL DEFAULT 0,    -- orden del admin, desempata a igualdad de votos
    votes_count    BIGINT  NOT NULL DEFAULT 0,
    created_at     TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at     TIMESTAMPTZ NOT NULL DEFAULT now(),
    CONSTRAINT chk_playlist_tracks_month_start  CHECK (EXTRACT(DAY FROM month_start) = 1),
    CONSTRAINT uq_playlist_tracks_month_spotify UNIQUE (month_start, spotify_id)
);


-- ============================================================================
--  PLAYLIST_VOTES — un usuario logueado vota una cancion (un voto por cancion)
-- ============================================================================
CREATE TABLE playlist_votes (
    user_id    UUID NOT NULL REFERENCES users(id)           ON DELETE CASCADE,
    track_id   UUID NOT NULL REFERENCES playlist_tracks(id) ON DELETE CASCADE,
    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    PRIMARY KEY (user_id, track_id)
);
CREATE INDEX idx_playlist_votes_track ON playlist_votes(track_id);

@Entity
@Table(name = "playlist_tracks")
@Getter
@Setter
public class PlaylistTracks {
    @Id
    @GeneratedValue(strategy = GenerationType.UUID)
    @Column(name = "id", columnDefinition = "uuid", updatable = false, nullable = false)

    private String title;

    @Column(name = "artist", nullable = false)

    private String spotifyId;

    /** Orden elegido por el admin: desempata cuando dos canciones tienen los mismos votos. */
    @Column(name = "sort_order", nullable = false)

package com.kryptosystems.ballastasera.repositories;

import com.kryptosystems.ballastasera.models.entities.PlaylistTracks;
import org.springframework.data.jpa.repository.JpaRepository;
import org.springframework.data.jpa.repository.Modifying;
import org.springframework.data.jpa.repository.Query;
import org.springframework.data.repository.query.Param;

import java.time.LocalDate;
import java.util.List;
import java.util.Optional;
import java.util.UUID;

public interface PlaylistTracksRepository extends JpaRepository<PlaylistTracks, UUID> {

    /** Mes de la lista vigente: el mas reciente que ya empezo. Vacio si todavia no hay ninguna lista. */
    @Query("SELECT MAX(t.monthStart) FROM PlaylistTracks t WHERE t.monthStart <= :today")
    Optional<LocalDate> findCurrentMonthStart(@Param("today") LocalDate today);

    /** Meses que tienen lista y ya empezaron, del mas reciente al mas antiguo. */
    @Query("SELECT DISTINCT t.monthStart FROM PlaylistTracks t WHERE t.monthStart <= :today ORDER BY t.monthStart DESC")
    List<LocalDate> findMonthStarts(@Param("today") LocalDate today);

    /** Canciones de un mes por votos; a igualdad de votos manda el orden del admin. */
    @Query("""
            SELECT t FROM PlaylistTracks t
            JOIN FETCH t.danceStyle
            WHERE t.monthStart = :monthStart
            ORDER BY t.votesCount DESC, t.sortOrder ASC, t.title ASC
            """)
    List<PlaylistTracks> findByMonthStartOrderedByVotes(@Param("monthStart") LocalDate monthStart);

    @Modifying
    @Query("UPDATE PlaylistTracks t SET t.votesCount = t.votesCount + 1 WHERE t.id = :trackId")
    void incrementVotesCount(@Param("trackId") UUID trackId);

    @Modifying
    @Query("UPDATE PlaylistTracks t SET t.votesCount = t.votesCount - 1 WHERE t.id = :trackId AND t.votesCount > 0")
    void decrementVotesCount(@Param("trackId") UUID trackId);

    @Modifying
    @Query("UPDATE PlaylistTracks t SET t.votesCount = t.votesCount - 1 WHERE t.id IN :trackIds AND t.votesCount > 0")
    void decrementVotesCountForTracks(@Param("trackIds") List<UUID> trackIds);
}

package com.kryptosystems.ballastasera.services.implementations;

import com.kryptosystems.ballastasera.exceptions.PlaylistVotingClosedException;
import com.kryptosystems.ballastasera.models.entities.PlaylistTracks;
import com.kryptosystems.ballastasera.models.entities.PlaylistVotes;
import com.kryptosystems.ballastasera.models.entities.keys.UserTrackId;
import com.kryptosystems.ballastasera.repositories.PlaylistTracksRepository;
import com.kryptosystems.ballastasera.repositories.PlaylistVotesRepository;
import com.kryptosystems.ballastasera.repositories.UsersRepository;
import com.kryptosystems.ballastasera.services.manager.PlaylistService;
import jakarta.persistence.EntityNotFoundException;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.time.LocalDate;
import java.time.YearMonth;
import java.util.List;
import java.util.Optional;
import java.util.UUID;

@Service
@RequiredArgsConstructor
public class PlaylistServiceImpl implements PlaylistService {

    private final PlaylistTracksRepository playlistTracksRepository;
    private final PlaylistVotesRepository playlistVotesRepository;
    private final UsersRepository usersRepository;

    @Override
    public List<LocalDate> findMonthStarts() {
        return playlistTracksRepository.findMonthStarts(LocalDate.now());
    }

    @Override
    public List<PlaylistTracks> findByMonth(YearMonth month) {
        if (month == null) {
            return findCurrentMonthStart()
                    .map(playlistTracksRepository::findByMonthStartOrderedByVotes)
                    .orElse(List.of());
        }
        LocalDate monthStart = month.atDay(1);
        /** Una lista cargada con antelacion no se ve hasta que empieza su mes. */
        if (monthStart.isAfter(LocalDate.now())) {
            return List.of();
        }
        return playlistTracksRepository.findByMonthStartOrderedByVotes(monthStart);
    }

    @Override
    public List<PlaylistVotes> findCurrentVotesByUserId(UUID userId) {
        return findCurrentMonthStart()
                .map(monthStart -> playlistVotesRepository.findByUserIdAndMonthStart(userId, monthStart))
                .orElse(List.of());
    }

    @Override
    @Transactional
    public void addVote(UUID userId, UUID trackId) {
        PlaylistTracks track = findVotableTrack(trackId);
        if (playlistVotesRepository.existsById(voteId(userId, trackId))) {
            return;
        }
        PlaylistVotes vote = new PlaylistVotes();
        vote.setUser(usersRepository.getReferenceById(userId));
        vote.setTrack(track);
        playlistVotesRepository.save(vote);
        playlistTracksRepository.incrementVotesCount(trackId);
    }

    @Override
    @Transactional
    public void removeVote(UUID userId, UUID trackId) {
        findVotableTrack(trackId);
        UserTrackId id = voteId(userId, trackId);
        if (playlistVotesRepository.existsById(id)) {
            playlistVotesRepository.deleteById(id);
            playlistTracksRepository.decrementVotesCount(trackId);
        }
    }

    /** La lista vigente es la del mes mas reciente que ya empezo: si un mes no se carga
     * lista nueva, sigue valiendo la anterior. */
    private Optional<LocalDate> findCurrentMonthStart() {
        return playlistTracksRepository.findCurrentMonthStart(LocalDate.now());
    }

    /** Solo se vota (o se quita el voto) en la lista vigente: los meses anteriores
     * quedan congelados con su ranking final. */
    private PlaylistTracks findVotableTrack(UUID trackId) {
        PlaylistTracks track = playlistTracksRepository.findById(trackId)
                .orElseThrow(() -> new EntityNotFoundException("Playlist track not found with id " + trackId));
        if (!track.getMonthStart().equals(findCurrentMonthStart().orElse(null))) {
            throw new PlaylistVotingClosedException(
                    "Voting is closed for the playlist of " + YearMonth.from(track.getMonthStart()));
        }
        return track;
    }

    private UserTrackId voteId(UUID userId, UUID trackId) {
        UserTrackId id = new UserTrackId();
        id.setUserId(userId);
        id.setTrackId(trackId);
        return id;
    }
}

package com.kryptosystems.ballastasera.models.mappers;

import org.mapstruct.Named;

import java.time.YearMonth;
import java.util.UUID;

@Mapper(componentModel = "spring")
public interface PlaylistMapper {

    @Mapping(target = "month", source = "monthStart", qualifiedByName = "toMonth")
    @Mapping(target = "danceStyle", source = "danceStyle.slug")

        return vote.getId().getTrackId();
    }

import org.springframework.security.core.annotation.AuthenticationPrincipal;
import org.springframework.web.bind.annotation.*;

import java.time.YearMonth;
import java.util.List;
import java.util.UUID;

import static com.kryptosystems.ballastasera.utilities.RestConstants.PLAYLIST;

@RestController
@RequestMapping(PLAYLIST)
@RequiredArgsConstructor
public class PlaylistController {

    private static final String GET_TRACKS = "";
    private static final String GET_MONTHS = "/months";
    private static final String GET_MY_VOTES = "/votes";
    private static final String ADD_VOTE = "/{id}/vote";
    private static final String REMOVE_VOTE = "/{id}/vote";

    private final PlaylistService playlistService;
    private final PlaylistMapper playlistMapper;

    /** Publico. Canciones de un mes (month=2026-10) ordenadas por votos.
     * Sin month devuelve la lista vigente; un mes sin lista devuelve una lista vacia. */
    @GetMapping(GET_TRACKS)
    public ResponseEntity<List<PlaylistTrackDto>> getTracks(@RequestParam(required = false) YearMonth month) {
        return ResponseEntity.ok(playlistService.findByMonth(month).stream()
                .map(playlistMapper::toPlaylistTrackDto)
                .toList());
    }

    /** Publico. Meses con lista (formato 2026-11), del mas reciente al mas antiguo.
     * El primero es el vigente, el unico en el que se puede votar. */
    @GetMapping(GET_MONTHS)
    public ResponseEntity<List<String>> getMonths() {
        return ResponseEntity.ok(playlistService.findMonthStarts().stream()
                .map(playlistMapper::toMonth)
                .toList());
    }

    /** Requiere estar autenticado. Ids de las canciones de la lista vigente que el usuario ya voto. */
    @GetMapping(GET_MY_VOTES)
    public ResponseEntity<List<UUID>> getMyVotes(@AuthenticationPrincipal UserPrincipal principal) {
        return ResponseEntity.ok(playlistService.findCurrentVotesByUserId(principal.getId()).stream()
                .map(playlistMapper::toTrackId)
                .toList());
    }

    /** Requiere estar autenticado. Vota una cancion de la lista vigente. Idempotente. */
    @PostMapping(ADD_VOTE)
    public ResponseEntity<Void> addVote(@AuthenticationPrincipal UserPrincipal principal, @PathVariable UUID id) {
        playlistService.addVote(principal.getId(), id);
        return ResponseEntity.noContent().build();
    }

    /** Requiere estar autenticado. Quita el voto. */
    @DeleteMapping(REMOVE_VOTE)
    public ResponseEntity<Void> removeVote(@AuthenticationPrincipal UserPrincipal principal, @PathVariable UUID id) {
        playlistService.removeVote(principal.getId(), id);
        return ResponseEntity.noContent().build();
    }
}
